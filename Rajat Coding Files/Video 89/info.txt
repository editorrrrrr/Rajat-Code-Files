ok so if you want to handle different server from one place doesnt matter where it created u just simply drag and drop this into the main and connect it with the express by using this


const express = require('express');
const router = express.Router();

// define the home page route
router.get('/', (req, res) => {
  res.send('blog home page');
});
// define the about route
router.get('/about', (req, res) => {
  res.send('About blog');
});
router.get('/blogpost/:slug', (req, res) => {
  res.send(`fetch the api for ${req.params.slug}`);
});

module.exports = router;
